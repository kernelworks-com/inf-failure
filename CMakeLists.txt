cmake_minimum_required(VERSION 3.20)
project(inf_failure VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(INF_FAILURE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(inf_failure src/replay.cpp)
target_include_directories(inf_failure PUBLIC include)
target_compile_features(inf_failure PUBLIC cxx_std_20)
set_target_properties(inf_failure PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(inf_failure PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(INF_FAILURE_SANITIZERS)
    target_compile_options(inf_failure PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(inf_failure PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(inf-failure src/main.cpp)
target_link_libraries(inf-failure PRIVATE inf_failure)
if(BUILD_TESTING)
  add_executable(replay_test tests/replay_test.cpp)
  target_link_libraries(replay_test PRIVATE inf_failure)
  add_test(NAME reference-replay COMMAND replay_test)
  add_test(NAME fresh-cli-replay COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:inf-failure> -DCASE_PATH=${CMAKE_CURRENT_BINARY_DIR}/roundtrip.iff
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.cmake)
endif()
